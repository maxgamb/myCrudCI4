<?php
/**
 * Frammento da integrare nel codice generato del Create.
 * Variabili attese: $relationalCreateConfig, $errors.
 */
?>
<?php foreach (($relationalCreateConfig ?? []) as $relationKey => $relation): ?>
    <?php $create = $relation['create'] ?? []; ?>
    <?php if (empty($relation['enabled']) || empty($create['enabled'])) continue; ?>

    <div class="card shadow-sm mt-4" data-relational-create="<?= esc($relationKey) ?>">
        <div class="card-header d-flex justify-content-between align-items-center">
            <strong><i class="bi bi-diagram-3"></i> <?= esc($relation['title'] ?? $relationKey) ?></strong>
            <button type="button" class="btn btn-sm btn-outline-primary" data-add-relation-row="<?= esc($relationKey) ?>">
                <i class="bi bi-plus-circle"></i> Aggiungi
            </button>
        </div>
        <div class="card-body">
            <div data-relation-rows="<?= esc($relationKey) ?>"></div>
            <template data-relation-template="<?= esc($relationKey) ?>">
                <div class="border rounded p-3 mb-3 relation-row">
                    <div class="row g-3">
                        <?php foreach (($create['fields'] ?? []) as $fieldName => $field): ?>
                            <?php
                            $fieldName = is_string($fieldName) ? $fieldName : (string) ($field['name'] ?? '');
                            if ($fieldName === '') continue;
                            if (!empty($field['primary']) || !empty($field['autoIncrement']) || !empty($field['databaseManaged'])) continue;
                            if ($fieldName === ($relation['foreignKey'] ?? '') || $fieldName === ($relation['primaryKey'] ?? '')) continue;
                            ?>
                            <div class="col-md-<?= max(1, min(12, (int) ($field['width'] ?? 6))) ?>">
                                <label class="form-label"><?= esc($field['label'] ?? $fieldName) ?></label>
                                <input type="<?= esc($field['inputType'] ?? 'text') ?>"
                                       class="form-control"
                                       data-field-name="<?= esc($fieldName) ?>">
                            </div>
                        <?php endforeach; ?>
                    </div>
                    <div class="text-end mt-3">
                        <button type="button" class="btn btn-sm btn-outline-danger" data-remove-relation-row>
                            <i class="bi bi-x-circle"></i> Rimuovi
                        </button>
                    </div>
                </div>
            </template>
        </div>
    </div>
<?php endforeach; ?>

<script>
document.addEventListener('DOMContentLoaded', () => {
    document.querySelectorAll('[data-relational-create]').forEach((card) => {
        const relationKey = card.dataset.relationalCreate;
        const rows = card.querySelector('[data-relation-rows]');
        const template = card.querySelector('[data-relation-template]');
        const addButton = card.querySelector('[data-add-relation-row]');

        let nextIndex = 0;

        const addRow = () => {
            const fragment = template.content.cloneNode(true);
            const row = fragment.querySelector('.relation-row');
            row.querySelectorAll('[data-field-name]').forEach((input) => {
                input.name = `relations[${relationKey}][${nextIndex}][${input.dataset.fieldName}]`;
            });
            nextIndex++;
            rows.appendChild(fragment);
        };

        addButton?.addEventListener('click', addRow);
        rows?.addEventListener('click', (event) => {
            const button = event.target.closest('[data-remove-relation-row]');
            if (button) button.closest('.relation-row')?.remove();
        });
    });
});
</script>
